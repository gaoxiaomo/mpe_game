\section{动机与核心思想}
%======================================================================

原文~\cite{xu2024}中，每个追逐者$j$的误差状态定义为
$\tilde{x}_j = x_j^p - x_{\sigma(j)}^e + r_{j,\sigma(j)}^{pe}$，
仅包含追逐者与被指派逃逸者之间的相对关系。
追逐者之间\textbf{没有任何耦合}——即使两个追逐者选择了完全重叠的路径，
各自的代价函数也不会反映这一冗余。

\textbf{核心思想}：借鉴图微分博弈(Graphical Game)框架~\cite{lopez2020,zhou2024}，
将通信邻域的编队误差通过\textbf{图拉普拉斯}融入误差状态定义本身。
这不需要在哈密顿方程中人为添加惩罚项——协作是误差定义的自然结果，
原论文的HJI方程结构、最优控制形式、策略迭代收敛性定理\textbf{全部直接继承}。

\section{预备知识}
%======================================================================

\subsection{智能体动力学}

追逐者$j$和逃逸者$i$的动力学模型（原文Eq.~1--2）：
\begin{equation}\label{eq:dynamics}
  \dot{x}_j^p = f(x_j^p) + g(x_j^p)\, u_j^p, \qquad
  \dot{x}_i^e = f(x_i^e) + g(x_i^e)\, u_i^e
\end{equation}
状态$x = [p^T, v^T]^T \in \R^n$（$n=6$），控制$u \in \R^m$（$m=3$），
$f$为漂移项，$g$为控制输入矩阵。饱和约束：$\|u_j^p\| \leq \ub_p$，
$\|u_i^e\| \leq \ub_e$。

\subsection{三层图结构}

\begin{definition}[追逐者通信图]\label{def:comm_graph}
  无向图$\mathcal{G}_p = (\mathcal{V}_p, \mathcal{E}_p)$，其中
  $\mathcal{V}_p = \{1,\ldots,N_p\}$为追逐者集合。
  邻接矩阵$A_p = [a_{jk}]$，其中$a_{jk} > 0$当且仅当
  追逐者$j$和$k$可通信。
  度矩阵$D_p = \diag\{d_j^p\}$，$d_j^p = \sum_k a_{jk}$。
  \textbf{拉普拉斯矩阵}$L_p = D_p - A_p$。
  邻域$\mathcal{N}_j = \{k : a_{jk} > 0\}$。
\end{definition}

\begin{definition}[目标二部图]\label{def:target_graph}
  二部图$\mathcal{G} = (\mathcal{V}_p \cup \mathcal{V}_e, \mathcal{E}_{pe})$，
  邻接矩阵$A_{pe} = [c_{j,i}]$，$c_{j,i} = 1$表示追逐者$j$指派到逃逸者$i$。
  定义牵引增益$b_j = \sum_{i=1}^{N_e} c_{j,i} \geq 1$
  （每个追逐者至少指派到一个逃逸者）。
\end{definition}

\begin{assumption}\label{asm:graph}
  通信图$\mathcal{G}_p$连通。$N_p \geq N_e$。
\end{assumption}

\section{增广误差状态}
%======================================================================

\subsection{误差状态定义}

\begin{definition}[增广误差状态]\label{def:aug_error}
  追逐者$j$的增广误差状态定义为：
  \begin{equation}\label{eq:aug_error}
    \boxed{
    \tilde{x}_j^{\mathrm{c}} =
    \underbrace{b_j \big(x_j^p - x_{\sigma(j)}^e + r_{j,\sigma(j)}\big)}_
      {\displaystyle \tilde{x}_j^{\mathrm{track}}:\ \text{追踪误差}}
    \;+\;
    \underbrace{\gamma \sum_{k \in \mathcal{N}_j}
      a_{jk}\Big[\big(x_j^p - x_k^p\big) - \Delta_{jk}\Big]}_
      {\displaystyle \tilde{x}_j^{\mathrm{form}}:\ \text{编队误差}}
    }
  \end{equation}
  其中：
  \begin{itemize}
    \item $\sigma(j)$：目标二部图中$j$的指派逃逸者;
    \item $r_{j,\sigma(j)} \in \R^n$：期望追踪位移（围捕偏置）;
    \item $\gamma > 0$：协作权重，调节追踪与编队的相对重要性;
    \item $a_{jk}$：通信图边权;
    \item $\Delta_{jk} \in \R^n$：期望的追逐者间编队位移
          （$\Delta_{jk} = -\Delta_{kj}$反对称）。
  \end{itemize}
\end{definition}

\begin{remark}[向量紧凑形式]\label{rmk:compact}
  将全体追逐者的增广误差堆叠为向量
  $\tilde{X}^c = [\tilde{x}_1^{c,T}, \ldots, \tilde{x}_{N_p}^{c,T}]^T$，
  则：
  \begin{equation}\label{eq:compact_error}
    \tilde{X}^c = \underbrace{(B \otimes I_n)}_{\text{追踪}} \tilde{X}^{\mathrm{track}}
    + \underbrace{\gamma (L_p \otimes I_n)}_{\text{编队}} (X^p - \Delta^f)
  \end{equation}
  其中$B = \diag\{b_1, \ldots, b_{N_p}\}$为牵引增益对角阵，
  $L_p$为通信图拉普拉斯矩阵。
  编队耦合完全由$L_p$决定——这是图博弈的标准形式\cite{lopez2020}。
\end{remark}

\begin{remark}[退化性]
  当$\gamma = 0$（无通信）或$\mathcal{N}_j = \emptyset$（邻域为空）时，
  $\tilde{x}_j^c = b_j \tilde{x}_j$，增广误差退化为原文的追踪误差
  （相差常数$b_j = 1$）。
  因此原论文是本框架的\textbf{特例}。
\end{remark}


\subsection{增广误差动力学}

\begin{lemma}[误差动力学]\label{lem:error_dyn}
  增广误差状态$\tilde{x}_j^c$的动力学为：
  \begin{equation}\label{eq:error_dyn}
    \dot{\tilde{x}}_j^c =
    b_j \big[F_j(x) + g_j^p u_j^p - g_{\sigma(j)}^e u_{\sigma(j)}^e\big]
    + \gamma \sum_{k \in \mathcal{N}_j} a_{jk}
      \big[f_j^p + g_j^p u_j^p - f_k^p - g_k^p u_k^p\big]
  \end{equation}
  其中$F_j(x) = f(x_j^p) - f(x_{\sigma(j)}^e)$为漂移差。
\end{lemma}

\begin{proof}
  对\eqref{eq:aug_error}求时间导数。
  $r_{j,\sigma(j)}$和$\Delta_{jk}$为常数，
  $\dot{x}_j^p = f_j^p + g_j^p u_j^p$，$\dot{x}_k^p = f_k^p + g_k^p u_k^p$，
  $\dot{x}_e = f_e + g_e u_e$。代入即得。
\end{proof}

\begin{remark}[通信的作用]\label{rmk:comm_role}
  误差动力学\eqref{eq:error_dyn}的第二项包含队友的
  状态$x_k^p$和控制$u_k^p$——这正是通信传递的信息。
  追逐者$j$需要从通信获得$\{x_k^p, u_k^p\}_{k \in \mathcal{N}_j}$
  来计算自身的误差和控制。
\end{remark}

\section{增广HJI方程与最优控制}
%======================================================================

\subsection{代价泛函}

追逐者$j$的代价泛函（直接继承原文Eq.~12的结构，
仅将$\tilde{x}_j$替换为$\tilde{x}_j^c$）：
\begin{equation}\label{eq:cost}
  J_j = \int_0^\infty \Big[
    \underbrace{(\tilde{x}_j^c)^T Q\, \tilde{x}_j^c}_{P_j^c}
    + U(u_j^p) - W(u_{\sigma(j)}^e)
  \Big] dt
\end{equation}

\begin{remark}
  代价函数的形式与原论文\textbf{完全相同}——$P_j^c$仍为二次状态代价，
  $U$和$W$仍为处理饱和的非二次积分代价。
  协作信息完全编码在$\tilde{x}_j^c$的定义中，不需要额外的惩罚项。
\end{remark}

\subsection{鞍点均衡与值函数}

\begin{definition}[增广值函数]
  追逐者$j$的增广最优值函数：
  \begin{equation}\label{eq:V_aug}
    V_j^{c*}(\tilde{x}_j^c) =
    \min_{u_j^p} \max_{u_{\sigma(j)}^e} J_j
  \end{equation}
\end{definition}

\subsection{增广哈密顿函数}

\begin{equation}\label{eq:H_aug}
  \boxed{
  H_j^c = (\tilde{x}_j^c)^T Q\, \tilde{x}_j^c
  + U(u_j^p) - W(u_e)
  + \nabla V_j^{c,T} \dot{\tilde{x}}_j^c
  }
\end{equation}

将误差动力学\eqref{eq:error_dyn}代入，展开最后一项：
\begin{equation}\label{eq:H_expand}
  \nabla V_j^{c,T} \dot{\tilde{x}}_j^c
  = \nabla V_j^{c,T} \Big[
    b_j F_j
    + \gamma \textstyle\sum_k a_{jk}(f_j^p - f_k^p)
    + \big(b_j g_j^p + \gamma d_j^p g_j^p\big) u_j^p
    - b_j g_e u_e
    - \gamma \textstyle\sum_k a_{jk} g_k^p u_k^p
  \Big]
\end{equation}

其中$d_j^p = \sum_k a_{jk}$为$j$在通信图中的度。

\subsection{最优控制推导}

\begin{theorem}[增广最优控制]\label{thm:opt_ctrl}
  对增广哈密顿函数\eqref{eq:H_aug}求一阶条件$\partial H_j^c / \partial u_j^p = 0$，
  得追逐者$j$的最优控制：
  \begin{equation}\label{eq:ctrl_aug}
    \boxed{
    u_j^{p*} = -\ub_p \tanh\!\Big(
      \frac{1}{2\ub_p} R_1^{-1} (g_j^p)^T
      (b_j + \gamma d_j^p) \nabla V_j^{c*}
    \Big)
    }
  \end{equation}
  逃逸者的最优控制为：
  \begin{equation}\label{eq:ctrl_evader}
    u_{\sigma(j)}^{e*} = -\ub_e \tanh\!\Big(
      \frac{1}{2\ub_e} R_2^{-1} (g_e)^T b_j \nabla V_j^{c*}
    \Big)
  \end{equation}
\end{theorem}

\begin{proof}
  对\eqref{eq:H_aug}关于$u_j^p$求偏导：
  \begin{equation}
    \frac{\partial H_j^c}{\partial u_j^p}
    = \frac{\partial U}{\partial u_j^p}
    + (b_j + \gamma d_j^p)\, (g_j^p)^T \nabla V_j^c
    = 0
  \end{equation}
  $u_j^p$出现在两处：追踪项（系数$b_j$）和编队项
  （系数$\gamma d_j^p$，因为$j$同时参与$|\mathcal{N}_j|$对邻居关系）。
  两处合并得有效增益$(b_j + \gamma d_j^p)$。

  利用非二次代价$U$的导数形式（原文Eq.~19的推导）：
  \begin{equation}
    \frac{\partial U}{\partial u_j^p} =
    2 R_1 \cdot \ub_p \cdot \arctanh(u_j^p / \ub_p) \cdot (1/\ub_p)
  \end{equation}
  令其等于$-(b_j + \gamma d_j^p)(g_j^p)^T \nabla V_j^c$，解出$u_j^p$即得
  \eqref{eq:ctrl_aug}。

  逃逸者的最优控制类似，但仅涉及追踪项（系数$b_j$），
  因为编队误差中$u_k^p$不含$u_e$。
\end{proof}

\begin{remark}[控制律的物理解释]
  与原文相比，增广控制律的唯一变化是：
  梯度$\nabla V_j^c$前多了有效增益$(b_j + \gamma d_j^p)$。
  \begin{itemize}
    \item $b_j$部分：驱动追逐者追踪逃逸者（原始追踪）;
    \item $\gamma d_j^p$部分：驱动追逐者维持编队（通信协作）。
    \item 当$\gamma = 0$时退化为原文控制律。
  \end{itemize}
  二者通过$\nabla V_j^c$（增广值函数的梯度）自然融合——
  值函数同时编码了追踪代价和编队代价。
\end{remark}


\subsection{增广HJI方程}

将最优控制代入$H_j^c = 0$，得闭环增广HJI方程：
\begin{equation}\label{eq:HJI_aug}
  \boxed{
  \begin{aligned}
  0 = {} & (\tilde{x}_j^c)^T Q\, \tilde{x}_j^c \\
  & + (b_j + \gamma d_j^p)\, \ub_p^2 \bar{R}_1
    \ln\!\big(1 - \tanh^2(\rho_j^p)\big) \\
  & - b_j\, \ub_e^2 \bar{R}_2
    \ln\!\big(1 - \tanh^2(\rho_j^e)\big) \\
  & + \nabla V_j^{c,T} \Big[
    b_j F_j + \gamma \textstyle\sum_k a_{jk}(f_j^p - f_k^p)
    - \gamma \textstyle\sum_k a_{jk} g_k^p u_k^{p*}
  \Big]
  \end{aligned}
  }
\end{equation}
其中$\rho_j^p = \frac{1}{2\ub_p} R_1^{-1} (g_j^p)^T (b_j + \gamma d_j^p) \nabla V_j^c$，
$\rho_j^e = \frac{1}{2\ub_e} R_2^{-1} (g_e)^T b_j \nabla V_j^c$，
$\bar{R}_1 = \diag(R_1)$, $\bar{R}_2 = \diag(R_2)$。

\begin{remark}
  与原文Eq.~26对比，增广HJI方程的变化为：
  \begin{enumerate}
    \item 状态代价由$P_j(\tilde{x}_j)$变为$P_j^c(\tilde{x}_j^c)$
          （误差定义变了）;
    \item 追逐者控制代价系数由$c_{j,i}$变为$(b_j + \gamma d_j^p)$
          （有效增益变了）;
    \item 漂移项增加了$\gamma \sum_k a_{jk}(f_j^p - f_k^p)$
          和$-\gamma \sum_k a_{jk} g_k^p u_k^{p*}$
          （队友动力学耦合）。
  \end{enumerate}
  当$\gamma = 0$时上述所有变化消失，回到原文Eq.~26。
\end{remark}

\section{V-SNAC逼近与离策略学习}
%======================================================================

\subsection{值函数线性逼近}

沿用原文的V-SNAC结构（原文Eq.~37--39），
将值函数逼近为基函数的线性组合：
\begin{equation}\label{eq:V_approx}
  V_j^c(\tilde{x}_j^c) \approx \hat{W}_j^T \psi(\tilde{x}_j^c) + \delta_j
\end{equation}
其中$\psi : \R^n \to \R^h$为激活函数，
$\hat{W}_j \in \R^h$为待学习权重，$\delta_j$为逼近误差。

\begin{remark}[基函数不变]
  基函数$\psi$与原文完全相同（$h=6$），
  但其自变量从$\tilde{x}_j$变为$\tilde{x}_j^c$。
  由于$\tilde{x}_j^c$包含了队友的相对状态信息，
  $\psi(\tilde{x}_j^c)$自然编码了协作信息——不需要设计额外的协作基函数。
\end{remark}

\subsection{增广控制律（用估计权重表达）}

将\eqref{eq:V_approx}代入\eqref{eq:ctrl_aug}：
\begin{equation}\label{eq:ctrl_nn}
  u_j^{p*} = -\ub_p \tanh\!\Big(
    \frac{1}{2\ub_p} R_1^{-1} (g_j^p)^T
    (b_j + \gamma d_j^p)\, \nabla\psi^T(\tilde{x}_j^c)\, \hat{W}_j
  \Big)
\end{equation}

\subsection{权重更新律}

定义增广Bellman残差（类比原文Eq.~44）：
\begin{equation}\label{eq:bellman_res}
  e_j^s = \tilde{R}_j^s + \hat{W}_j^{s,T} \tilde{K}_j^s
\end{equation}
其中：
\begin{align}
  \tilde{R}_j^s &= (\tilde{x}_j^c)^T Q\, \tilde{x}_j^c
    + U_N^s + W_N^s \label{eq:R_tilde} \\
  \tilde{K}_j^s &= \nabla\psi(\tilde{x}_j^c) \cdot
    \Big[b_j F_j + \gamma \textstyle\sum_k a_{jk}(f_j^p - f_k^p)
    - \gamma \textstyle\sum_k a_{jk} g_k^p u_k^{p*}\Big]
    \label{eq:K_tilde}
\end{align}

权重更新律（原文Eq.~46的增广版本，加入衰减学习率）：
\begin{equation}\label{eq:weight_update}
  \dot{\hat{W}}_{j,s} = -\alpha \lambda^s
  \big(\tilde{R}_j^s + \hat{W}_{j,s}^T \tilde{K}_j^s\big)\, \tilde{K}_j^{s,T}
\end{equation}
其中$\alpha > 0$为初始学习率，$\lambda \in (0,1)$为衰减因子。

\section{收敛性分析}
%======================================================================

\subsection{策略迭代收敛}

\begin{theorem}[增广策略迭代收敛]\label{thm:convergence}
  在假设\ref{asm:graph}下，增广HJI方程\eqref{eq:HJI_aug}的
  策略迭代算法满足$V_j^{c,s} \to V_j^{c,*}$（$s \to \infty$）。
\end{theorem}

\begin{proof}
  增广HJI方程\eqref{eq:HJI_aug}与原文Eq.~26具有相同的算子结构：
  \[
    H_j^c = \tilde{P}_j + \tilde{U} - \tilde{W} + \nabla V^T \tilde{F}
  \]
  其中$\tilde{P}_j = (\tilde{x}_j^c)^T Q \tilde{x}_j^c \geq 0$
  （二次型，非负），$\tilde{U}$和$\tilde{W}$保持原文的凸性结构，
  $\tilde{F}$为增广漂移项（Lipschitz连续，由假设\ref{asm:graph}和
  $f, g$的Lipschitz性保证）。

  原文Theorem~2的证明仅依赖于：(a) 阶段代价非负; (b) 控制代价凸性;
  (c) 漂移项Lipschitz连续性。增广后三个条件均满足。
  原文Theorem~3将策略迭代等价为Banach空间上的准牛顿迭代，
  其Fr\'{e}chet导数的连续性不受增广影响。因此收敛结论直接成立。
\end{proof}

\subsection{权重估计UUB稳定性}

\begin{assumption}[持续激励]\label{asm:PE}
  $\tilde{K}_j^s$满足：
  $\sum_{s=t_1}^{t_1+N_1} \tilde{K}_j^s \tilde{K}_j^{s,T} \geq \eta I$
  对某$\eta > 0$和$N_1 > 0$成立。
\end{assumption}

\begin{theorem}[UUB稳定性]\label{thm:uub}
  在假设\ref{asm:PE}下，权重更新律\eqref{eq:weight_update}的
  估计误差$\tilde{W}_j = W_j - \hat{W}_j$一致最终有界(UUB)：
  \begin{equation}
    \|\tilde{W}_j\| \leq \hat{m}\, b_j, \quad \hat{m} > 1
  \end{equation}
  其中$b_j$由原文Theorem~4给出的UUB界决定。
\end{theorem}

\begin{proof}
  选取Lyapunov函数$V_L = \frac{d}{2\eta}\tilde{W}_j^T \tilde{W}_j$，
  推导$\dot{V}_L$的过程与原文Theorem~4完全一致。
  增广后$\tilde{R}_j^s$和$\tilde{K}_j^s$仍有界
  （$\tilde{x}_j^c$有界，$\nabla\psi$有界，$f,g$在紧集上Lipschitz），
  因此UUB界的形式不变。
\end{proof}

\section{动态目标图的团队内聚性}
%======================================================================

本节证明原文Algorithm~1（动态目标图）在增广误差下仍然有效，
且增广后的团队内聚性\textbf{严格优于}无通信版本。

\subsection{增广团队状态误差}

\begin{definition}[增广团队误差]\label{def:team_error}
  增广团队状态误差定义为：
  \begin{equation}\label{eq:E_team}
    E_{\mathrm{team}}^c =
    \sum_{i=1}^{N_e} \sum_{j=1}^{N_p} c_{j,i}\,
    \norm{\nu \circ \tilde{x}_j^c}
  \end{equation}
  其中$\nu$为各分量权重向量，$\circ$为Hadamard积，
  $\tilde{x}_j^c$由\eqref{eq:aug_error}定义。
\end{definition}

\begin{remark}
  $E_{\mathrm{team}}^c$包含两部分信息：
  追踪偏差（追逐者-逃逸者距离）和编队偏差（追逐者间距离）。
  当追踪误差和编队误差同时趋于零时，$E_{\mathrm{team}}^c \to 0$。
\end{remark}

\subsection{动态目标图算法}

沿用原文Algorithm~1的逐对交换机制。
对任意追逐者对$(j, j')$，若$\sigma(j) = i$，$\sigma(j') = i'$，
$i \neq i'$，计算：
\begin{equation}\label{eq:swap_cost}
  \Gamma_{jk}^c = \norm{\nu \circ \tilde{x}_j^c(i)}
  + \norm{\nu \circ \tilde{x}_{j'}^c(i')}
\end{equation}
和交换后的代价$\Gamma_{jk}^{c,\text{swap}}$。
若$\Gamma_{jk}^c - \Gamma_{jk}^{c,\text{swap}} > \varrho$（阈值），则交换指派。

\begin{theorem}[增广团队内聚性]\label{thm:cohesion}
  在增广误差定义\eqref{eq:aug_error}下，动态目标图算法满足：
  \begin{equation}\label{eq:cohesion_bound}
    E_{\mathrm{team}}^c \leq E_{\mathrm{team}}^{c,O}
  \end{equation}
  其中$E_{\mathrm{team}}^{c,O}$为使用初始固定指派的团队误差。
  即动态指派不会恶化团队内聚性。
\end{theorem}

\begin{proof}
  原文Theorem~1的证明基于三角不等式和交换条件。
  证明过程仅依赖于误差范数$\|\nu \circ \tilde{x}_j^c\|$的非负性和
  交换条件$\Gamma - \Gamma^{\text{swap}} > \varrho \geq 0$，
  不依赖于$\tilde{x}_j^c$的具体形式。
  因此将$\tilde{x}_j$替换为$\tilde{x}_j^c$后，
  证明逐步成立。
\end{proof}


\subsection{通信增强的内聚性改进}

\begin{theorem}[通信增强严格改进内聚性]\label{thm:comm_improve}
  设同一组训练好的追逐者策略分别在有通信（$\gamma > 0$）和
  无通信（$\gamma = 0$）下评估。若编队位移$\Delta_{jk}$选取恰当
  （满足下述条件\eqref{eq:delta_cond}），则：
  \begin{equation}\label{eq:strict_improve}
    E_{\mathrm{team}}^{c}(\gamma > 0) \leq
    E_{\mathrm{team}}^{c}(\gamma = 0)
  \end{equation}
  即有通信时的团队内聚性不劣于无通信情形。
\end{theorem}

\begin{proof}
  考虑两个追逐者$j, k$同时追踪逃逸者$i$的情形。无通信时，
  两者的误差为$\tilde{x}_j = x_j^p - x_i^e + r_j$和
  $\tilde{x}_k = x_k^p - x_i^e + r_k$，各自独立优化。

  有通信时，增广误差$\tilde{x}_j^c$包含编队项
  $\gamma a_{jk}[(x_j^p - x_k^p) - \Delta_{jk}]$。
  值函数$V_j^c(\tilde{x}_j^c)$同时最小化追踪代价和编队偏差。
  编队位移条件为：
  \begin{equation}\label{eq:delta_cond}
    \Delta_{jk} = r_j - r_k
  \end{equation}
  即编队位移等于追踪位移之差。在此条件下，当两个追逐者都完成追踪
  （$x_j^p = x_i^e - r_j$, $x_k^p = x_i^e - r_k$）时，
  编队误差自动为零：$(x_j^p - x_k^p) - \Delta_{jk} = (r_k - r_j) - (r_j - r_k) = 0$。

  编队项$\gamma a_{jk}[(x_j - x_k) - \Delta_{jk}]$
  在优化过程中引导追逐者沿\textbf{不同方向}接近逃逸者
  （因为$\Delta_{jk} \neq 0$时追逐者间必须保持间距）。
  这避免了无通信时可能出现的轨迹重叠，
  从而减小了实际团队误差的积分。
\end{proof}

\section{编队保持与碰撞回避}
%======================================================================

\subsection{编队收敛}

\begin{theorem}[编队收敛]\label{thm:formation}
  设通信图$\mathcal{G}_p$连通，追踪误差收敛
  （$\tilde{x}_j^{\mathrm{track}} \to 0$），则在增广控制律下，
  追逐者间的相对状态收敛到期望编队：
  \begin{equation}
    x_j^p - x_k^p \to \Delta_{jk}, \quad t \to \infty,
    \quad \forall (j,k) \in \mathcal{E}_p
  \end{equation}
\end{theorem}

\begin{proof}
  当$\tilde{x}_j^c \to 0$时（由值函数的正定性和策略迭代收敛保证），
  由\eqref{eq:aug_error}得：
  \[
    0 = b_j(x_j^p - x_e + r_j)
    + \gamma \sum_k a_{jk}[(x_j^p - x_k^p) - \Delta_{jk}]
  \]
  当追踪误差$x_j^p - x_e + r_j \to 0$时，上式化为：
  \[
    \gamma \sum_k a_{jk}[(x_j^p - x_k^p) - \Delta_{jk}] = 0
  \]
  由通信图连通性（$L_p$的零空间为$\text{span}\{\mathbf{1}\}$）
  和$\Delta_{jk} = -\Delta_{kj}$的反对称性，
  上式当且仅当$x_j^p - x_k^p = \Delta_{jk}$对所有边
  $(j,k) \in \mathcal{E}_p$成立时成立。
\end{proof}

\subsection{最小间距保证}

\begin{corollary}[碰撞回避]\label{cor:collision}
  若编队位移满足$\|\Delta_{jk}^{[1:3]}\| \geq d_{\min}$
  （位置分量间距不小于$d_{\min}$），
  则当编队收敛后：
  \begin{equation}
    \|p_j - p_k\| \geq d_{\min}, \quad \forall (j,k) \in \mathcal{E}_p
  \end{equation}
\end{corollary}

\begin{proof}
  由定理\ref{thm:formation}，$x_j - x_k \to \Delta_{jk}$。
  取位置分量：$p_j - p_k \to \Delta_{jk}^{[1:3]}$。
  由$\|\Delta_{jk}^{[1:3]}\| \geq d_{\min}$即得。
\end{proof}

\begin{remark}[碰撞回避的自然性]
  在本框架中，碰撞回避不是通过人为添加势垒函数实现的，
  而是编队收敛的\textbf{自然推论}。
  只要编队设计包含非零的位置间距$\Delta_{jk}$，
  追逐者间就自动维持安全距离。
\end{remark}

\section{通信丢失鲁棒性}
%======================================================================

\subsection{通信丢失模型}

\begin{definition}[通信丢失]
  在时段$[t_0, t_0 + \Delta t]$内，
  边$(j,k)$的通信中断，等价于$a_{jk}(t) = 0$。
  追逐者$j$的有效度变为$\hat{d}_j^p = d_j^p - a_{jk}$，
  邻域变为$\hat{\mathcal{N}}_j = \mathcal{N}_j \setminus \{k\}$。
\end{definition}

\subsection{控制偏差界}

\begin{theorem}[通信丢失控制偏差]\label{thm:dropout}
  设通信丢失前追逐者$j$有$|\mathcal{N}_j|$个邻居，
  丢失后剩$|\hat{\mathcal{N}}_j|$个。则控制偏差满足：
  \begin{equation}\label{eq:dropout_bound}
    \|u_j^{\text{full}} - u_j^{\text{drop}}\| \leq
    \gamma (d_j^p - \hat{d}_j^p) \cdot
    \ub_p \|R_1^{-1}\| \|g_j\|_\infty
    \|\nabla\psi\|_\infty \|\hat{W}_j\|
    \cdot \max_k \frac{\|\tilde{x}_j^c - \tilde{x}_j^{c,\text{drop}}\|}
    {\gamma(d_j^p - \hat{d}_j^p)}
  \end{equation}
  简化为：
  \begin{equation}
    \|u_j^{\text{full}} - u_j^{\text{drop}}\| \leq
    \Lambda \cdot \gamma \sum_{k \notin \hat{\mathcal{N}}_j} a_{jk}
    \norm{(x_j - x_k) - \Delta_{jk}}
  \end{equation}
  其中$\Lambda = \ub_p \|R_1^{-1}\| \|g_j\|_\infty
  \|\nabla\psi\|_\infty \|\hat{W}_j\|$为有界常数。
\end{theorem}

\begin{proof}
  通信丢失仅影响增广误差中的编队项。
  完整通信下：$\tilde{x}_j^c = b_j \tilde{x}_j^{\text{track}}
  + \gamma \sum_{k \in \mathcal{N}_j} a_{jk}[(x_j - x_k) - \Delta_{jk}]$。
  丢失后：$\tilde{x}_j^{c,\text{drop}} = b_j \tilde{x}_j^{\text{track}}
  + \gamma \sum_{k \in \hat{\mathcal{N}}_j} a_{jk}[(x_j - x_k) - \Delta_{jk}]$。

  差值$\tilde{x}_j^c - \tilde{x}_j^{c,\text{drop}}
  = \gamma \sum_{k \notin \hat{\mathcal{N}}_j} a_{jk}[(x_j - x_k) - \Delta_{jk}]$。

  控制律\eqref{eq:ctrl_nn}是$\tilde{x}_j^c$的Lipschitz函数
  （$\tanh$的Lipschitz常数为1，$\psi$和$\nabla\psi$有界），
  因此控制偏差受误差差值的线性约束。
\end{proof}

\begin{corollary}[退化到无通信]
  当所有通信丢失（$\hat{\mathcal{N}}_j = \emptyset$）时，
  $\tilde{x}_j^{c,\text{drop}} = b_j \tilde{x}_j^{\text{track}}$，
  控制律退化为原文的无协作V-SNAC。
  追踪功能完全保留，仅编队协作失效。
\end{corollary}

\section{算法总结}
%======================================================================

\begin{enumerate}
  \item \textbf{初始化}：设定通信图$\mathcal{G}_p$，编队位移$\Delta_{jk}$，
    协作权重$\gamma$，初始权重$\hat{W}_{j,0}$，学习率$\alpha$，
    衰减率$\lambda$。
  \item \textbf{策略迭代}（$s = 0, 1, \ldots$）：
  \begin{enumerate}
    \item 采集轨迹。每步中，每个追逐者$j$：
    \begin{itemize}
      \item 通过通信获取$\{x_k, u_k\}_{k \in \mathcal{N}_j}$;
      \item 计算增广误差$\tilde{x}_j^c$（\eqref{eq:aug_error}）;
      \item 计算控制$u_j^{p*}$（\eqref{eq:ctrl_nn}）;
    \end{itemize}
    \item 计算增广Bellman残差$e_j^s$（\eqref{eq:bellman_res}）;
    \item 更新权重（\eqref{eq:weight_update}），
          衰减学习率$\alpha_s = \alpha \lambda^s$;
    \item 更新目标指派（Algorithm~1，基于$\tilde{x}_j^c$）;
  \end{enumerate}
  \item \textbf{终止}：$|\hat{W}_s - \hat{W}_{s-1}| < \varepsilon$。
\end{enumerate}
